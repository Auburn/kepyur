cmake_minimum_required(VERSION 3.22)
project(kepyur C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(KEPYUR_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
option(KEPYUR_RADIO "Build the radio (wifi-sdr + devourer + libusb)" ON)

# ---- DSP: no hardware, shared by the app and the desktop tools ----
add_library(kepyur_dsp STATIC dsp/video.cpp dsp/spectrum.cpp dsp/log.cpp)
target_include_directories(kepyur_dsp PUBLIC dsp)
target_compile_options(kepyur_dsp PRIVATE -O3 -ffast-math -Wall -Wextra)
if(ANDROID)
  target_link_libraries(kepyur_dsp PUBLIC log)
endif()

# ---- the picture on screen: an OpenGL ES 3 renderer (the app's Surface; off screen in kp_render) ----
if(ANDROID)
  add_library(kepyur_render STATIC render/video_renderer.cpp render/no_signal.cpp)
  target_link_libraries(kepyur_render PUBLIC kepyur_dsp GLESv3 EGL)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(KP_GLES IMPORTED_TARGET QUIET egl glesv2)  # e.g. Mesa: libegl-dev libgles-dev
  endif()
  if(KP_GLES_FOUND)
    add_library(kepyur_render STATIC render/video_renderer.cpp render/no_signal.cpp)
    target_link_libraries(kepyur_render PUBLIC kepyur_dsp PkgConfig::KP_GLES)
  endif()
endif()
if(TARGET kepyur_render)
  target_include_directories(kepyur_render PUBLIC render)
  target_compile_options(kepyur_render PRIVATE -O2 -Wall -Wextra)
endif()

# ---- radio ----
if(KEPYUR_RADIO)
  if(ANDROID)
    include(cmake/libusb_android.cmake)  # libusb from source + a pkg-config file devourer can find
  endif()
  # wifi-sdr assembles the on-chip streamer with mipsel binutils (shell.nix: KEPYUR_MIPS_PREFIX)
  if(DEFINED ENV{KEPYUR_MIPS_PREFIX})
    set(MIPS_AS "$ENV{KEPYUR_MIPS_PREFIX}as" CACHE FILEPATH "")
    set(MIPS_LD "$ENV{KEPYUR_MIPS_PREFIX}ld" CACHE FILEPATH "")
    set(MIPS_OBJCOPY "$ENV{KEPYUR_MIPS_PREFIX}objcopy" CACHE FILEPATH "")
  endif()
  include(cmake/patches.cmake)  # Play RX's wifi-sdr + devourer patches (patches/), applied once
  add_subdirectory(${KEPYUR_ROOT}/third_party/wifi-sdr wifi-sdr EXCLUDE_FROM_ALL)
  if(ANDROID)
    target_link_libraries(devourer PUBLIC usb-1.0-static)
  endif()
  add_library(kepyur_radio STATIC radio/radio.cpp radio/combine.cpp)
  target_include_directories(kepyur_radio PUBLIC radio)
  target_link_libraries(kepyur_radio PUBLIC kepyur_dsp wsdrcore)
  target_compile_options(kepyur_radio PRIVATE -O2 -Wall -Wextra)
endif()

# ---- desktop tools ----
if(NOT ANDROID)
  find_package(Threads REQUIRED)
  add_executable(kp_replay tools/kp_replay.cpp)
  target_link_libraries(kp_replay PRIVATE kepyur_dsp)
  target_compile_options(kp_replay PRIVATE -O2 -Wall)
  add_executable(kp_gen tools/kp_gen.cpp)  # a synthetic transmitter: recordings to test with, no hardware
  target_compile_options(kp_gen PRIVATE -O2 -Wall)
  if(TARGET kepyur_render)
    add_executable(kp_render tools/kp_render.cpp)  # the app's renderer off screen (needs EGL + OpenGL ES 3)
    target_link_libraries(kp_render PRIVATE kepyur_render)
    target_compile_options(kp_render PRIVATE -O2 -Wall)
  endif()
  if(KEPYUR_RADIO)
    add_executable(kp_live tools/kp_live.cpp)
    target_link_libraries(kp_live PRIVATE kepyur_radio Threads::Threads)
    target_compile_options(kp_live PRIVATE -O2 -Wall)
    # kp_live with a simulated receiver that plays a recording back as a tunable adapter (KP_SIM_FILE):
    # the scan -> pick -> watch flow end to end without hardware. sim_receiver.o supplies every
    # wsdr::Receiver symbol, so the archive's receiver.o is never pulled in.
    add_executable(kp_sim tools/kp_live.cpp radio/radio.cpp radio/combine.cpp tools/sim_receiver.cpp)
    target_include_directories(kp_sim PRIVATE radio)
    target_link_libraries(kp_sim PRIVATE kepyur_dsp wsdrcore Threads::Threads)
    target_compile_options(kp_sim PRIVATE -O2 -Wall)
  endif()
endif()

# ---- Android JNI ----
if(ANDROID)
  add_library(kepyur SHARED jni/kepyur_jni.cpp)
  target_link_libraries(kepyur PRIVATE kepyur_radio kepyur_render android log)
  target_compile_options(kepyur PRIVATE -O2 -Wall)
endif()
